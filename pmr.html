Integrasi Google Spreadsheet

Simpan otomatis data PDCA & CAPA langsung ke Google Sheets

✓ Terpasang di Kode
• Terhubung Langsung

Petunjuk Penggunaan Google Apps Script:

  1. Buka lembar kerja Google Sheets Anda.
  2. Klik menu Extensions > Apps Script.
  3. Tempelkan kode Apps Script berikut dan klik Deploy > New Deployment > Web app (akses: Anyone).
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Sheet 1: CAPA_Tracker
    var capaSheet = ss.getSheetByName("CAPA_Tracker") || ss.insertSheet("CAPA_Tracker");
    if (capaSheet.getLastRow() === 0) {
      capaSheet.appendRow(["ID", "Kategori", "Tipe", "Uraian Masalah", "Uraian Solusi", "Dept/PIC", "Due Date", "Status", "Ada Foto?"]);
    }
    if (data.capaList && data.capaList.length > 0) {
      capaSheet.clearContents();
      capaSheet.appendRow(["ID", "Kategori", "Tipe", "Uraian Masalah", "Uraian Solusi", "Dept/PIC", "Due Date", "Status", "Ada Foto?"]);
      data.capaList.forEach(function(item) {
        capaSheet.appendRow([item.id, item.category, item.type, item.problem, item.solution, item.pic, item.dueDate, item.status, item.evidence ? "Ya" : "Tidak"]);
      });
    }

    // Sheet 2: PDCA_Log
    var pdcaSheet = ss.getSheetByName("PDCA_Log") || ss.insertSheet("PDCA_Log");
    if (pdcaSheet.getLastRow() === 0) {
      pdcaSheet.appendRow(["No", "Tahap", "Aktivitas", "PIC/Dept", "Tanggal", "Output"]);
    }
    if (data.pdcaList && data.pdcaList.length > 0) {
      pdcaSheet.clearContents();
      pdcaSheet.appendRow(["No", "Tahap", "Aktivitas", "PIC/Dept", "Tanggal", "Output"]);
      data.pdcaList.forEach(function(item, idx) {
        pdcaSheet.appendRow([idx + 1, item.stage, item.title, item.pic, item.date, item.output]);
      });
    }

    return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Data tersimpan"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
