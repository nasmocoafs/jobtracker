Simpan otomatis data PDCA & CAPA langsung ke Google Sheets Anda

Status: Belum Terhubung
Panduan Cepat Menghubungkan Google Sheets (2 Menit):

1. Buka spreadsheet baru di Google Sheets (sheets.new).

2. Klik menu Extensions (Ekstensi) → Apps Script.

3. Hapus semua kode bawaan, lalu tempel (paste) kode Apps Script di bawah.

4. Klik tombol biru Deploy (Terapkan) di kanan atas → pilih New deployment (Penerapan baru) → jenis: Web app.

5. Pada opsi Who has access (Siapa yang memiliki akses), pilih: Anyone (Siapa saja) → Klik Deploy.

6. Salin Web App URL yang dihasilkan, lalu tempel pada isian di atas.

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Simpan Sheet CAPA
    var capaSheet = ss.getSheetByName("CAPA_Tracker") || ss.insertSheet("CAPA_Tracker");
    capaSheet.clearContents();
    capaSheet.appendRow(["ID", "Kategori", "Jenis", "Rencana Tindakan", "PIC / Dept", "Due Date", "Status", "Ada Bukti Foto"]);
    if (contents.capaList && contents.capaList.length > 0) {
      contents.capaList.forEach(function(item) {
        capaSheet.appendRow([
          item.id,
          item.category,
          item.type,
          item.plan,
          item.pic,
          item.dueDate,
          item.status,
          item.evidence ? "Ya (Terlampir)" : "-"
        ]);
      });
    }

    // 2. Simpan Sheet PDCA
    var pdcaSheet = ss.getSheetByName("PDCA_Log") || ss.insertSheet("PDCA_Log");
    pdcaSheet.clearContents();
    pdcaSheet.appendRow(["No", "Tahap", "Agenda / Aktivitas", "PIC / Dept", "Tanggal", "Output Hasil"]);
    if (contents.pdcaList && contents.pdcaList.length > 0) {
      contents.pdcaList.forEach(function(item, idx) {
        pdcaSheet.appendRow([idx + 1, item.step, item.agenda, item.pic, item.date, item.output]);
      });
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success", timestamp: new Date() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
